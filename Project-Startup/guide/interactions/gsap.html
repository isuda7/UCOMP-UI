지속 시간 duration
지연 시간 delay
가속도 효과 ease

반복 딜레이 repeatDelay
되돌아갈지 여부 yoyo
순차적으로 stagger

// Quickmenu A const qnbItems = document.querySelectorAll(".ex-qnb-a .qnb-item:not(:first-child)"); const qnbToggle = document.querySelector(".ex-qnb-a .qnb-item:first-child .qnb-link"); const qnbTl = gsap.timeline({ paused: true, reversed: true }); qnbTl.to(qnbItems, { y: (i) => -54 * (i + 1), opacity: 1, stagger: 0.015, duration: 0.3, pointerEvent: 'auto', ease: "power3.out" }); qnbToggle.addEventListener("click", () => { qnbTl.reversed() ? qnbTl.play() : qnbTl.reverse(); });
