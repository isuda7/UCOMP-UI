// GNB const gnb = document.querySelector('.gnb'); const pageHeader = document.querySelector('.page-header'); const gnbLinks = gnb.querySelectorAll('.dep1-link'); let openTimer; let closeTimer; function openFullDown(delay = 0) { clearTimeout(closeTimer); clearTimeout(openTimer); openTimer = setTimeout(() => { pageHeader.classList.add('is-active'); gnbLinks.forEach(link => link.setAttribute('aria-expanded', 'true')); }, delay); } function closeFullDown(delay = 200) { clearTimeout(openTimer); clearTimeout(closeTimer); closeTimer = setTimeout(() => { pageHeader.classList.remove('is-active'); gnbLinks.forEach(link => link.setAttribute('aria-expanded', 'false')); }, delay); } // 마우스: 딜레이 열기/닫기 gnb.addEventListener('mouseenter', () => openFullDown(200)); gnb.addEventListener('mouseleave', () => closeFullDown(200)); // 포커스: 즉시 열기 / 딜레이 닫기 gnb.addEventListener('focusin', () => openFullDown(0)); gnb.addEventListener('focusout', () => closeFullDown(200));
